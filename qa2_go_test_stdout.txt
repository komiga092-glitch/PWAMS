        'care_provided', 'loans', 'loan_repayments', 'revenue_records',
        'audit_logs'

    partner_role_id UUID;

    live_partner_users INT;

    SELECT id INTO partner_role_id FROM roles WHERE name = 'Partner' LIMIT 1;
